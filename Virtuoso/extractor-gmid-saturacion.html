Extractor automático de gm/Id y margen de saturación

Plugin en SKILL para ADE Assembler (Maestro) que detecta automáticamente los transistores de un esquemático — incluso dentro de jerarquías anidadas — y genera las expresiones de gm/Id y el margen de saturación para cada uno.

Qué hace Qué necesitas configurar
gm_id_plugin.il 
;; =========================================================================
;; PLUGIN: DETECTOR DE TRANSISTORES Y GENERADOR DE ECUACIONES GM/ID
;; Recorre jerarquia (hasta gmidMaxDepth niveles) desde el esquematico activo
;; =========================================================================

unless( boundp('gmidTests) gmidTests = nil )
unless( boundp('gmidSatMargin) gmidSatMargin = "0" )
unless( boundp('gmidMaxDepth) gmidMaxDepth = 3 )

;; -------------------------------------------------------------------------
;; Tabla de vistas concretas para celdas cuyo nombre de vista real no es
;; "schematic" a secas (p.ej. porque usas varias variantes segun el config).
;; Edita esto a mano segun que config estes usando en cada momento:
;;   gmidViewOverrides = list(cons("BufferSortida" "schematic_RailToRail"))
;; -------------------------------------------------------------------------
unless( boundp('gmidViewOverrides)
    gmidViewOverrides = list(cons("BufferSortida" "schematic_RailToRail"))
)

procedure( esTransistor(cellName)
    let( (cn)
        cn = lowerCase(or(cellName ""))
        (member(cn '("ne" "nmos" "nch" "nfet" "pe" "pmos" "pch" "pfet")) != nil)
    )
)

procedure( esTransistorN(cellName)
    let( (cn)
        cn = lowerCase(or(cellName ""))
        (member(cn '("ne" "nmos" "nch" "nfet")) != nil)
    )
)

;; Devuelve el nombre de vista a usar para una celda: el override si existe,
;; si no "schematic" por defecto.
procedure( gmidResolverVista(cellName)
    let( (ov)
        ov = assoc(cellName gmidViewOverrides)
        if( ov then cdr(ov) else "schematic")
    )
)

;; Devuelve lista de pares (fullPath isNMOS)
procedure( RecorrerJerarquiaGmId(cv path depth maxDepth)
    let( (resultado fullPath subCvRes subCv vName)
        resultado = nil
        when( cv && cv~>instances && depth <= maxDepth
            foreach( inst cv~>instances
                fullPath = if( path == "" then inst~>name else sprintf(nil "%s/%s" path inst~>name))
                if( esTransistor(inst~>cellName) then
                    resultado = cons(list(fullPath esTransistorN(inst~>cellName)) resultado)
                else
                    vName = gmidResolverVista(inst~>cellName)
                    subCvRes = errset(dbOpenCellViewByType(inst~>libName inst~>cellName vName nil "r"))
                    subCv = if( subCvRes then car(subCvRes) else nil)
                    when( subCv
                        resultado = append(RecorrerJerarquiaGmId(subCv fullPath depth+1 maxDepth) resultado)
                        dbClose(subCv)
                    )
                )
            )
        )
        resultado
    )
)

procedure( GenerarMaestroGmId()
    let( (cv sess tests xtors nN nP n1 n2 r1 r2 r3 nErr)
        cv = geGetEditCellView()
        when( cv && cv~>cellViewType == "schematic"
            sess = car(maeGetSessions())
            unless( sess
                sess = maeOpenSetup(cv~>libName cv~>cellName "maestro" ?mode "a"))

            tests = if( gmidTests
                then setof(tn maeGetSetup(?session sess) member(tn gmidTests))
                else maeGetSetup(?session sess))

            xtors = nil
            nN = 0
            nP = 0
            foreach( par RecorrerJerarquiaGmId(cv "" 1 gmidMaxDepth)
                xtors = cons(car(par) xtors)
                if( cadr(par) then nN = nN + 1 else nP = nP + 1 )
            )

            nErr = 0
            foreach( tst tests
                foreach( nm xtors
                    n1 = sprintf(nil "gmid_%s" nm)
                    n2 = sprintf(nil "vdsmargin_%s" nm)
                    errset(maeDeleteOutput(n1 tst ?session sess))
                    errset(maeDeleteOutput(n2 tst ?session sess))
                    r1 = maeAddOutput(n1 tst ?outputType "point"
                        ?expr sprintf(nil "OP(\"/%s\" \"gm\") / OP(\"/%s\" \"id\")" nm nm)
                        ?session sess)
                    r2 = maeAddOutput(n2 tst ?outputType "point"
                        ?expr sprintf(nil "abs(OP(\"/%s\" \"vds\")) - abs(OP(\"/%s\" \"vdsat\"))" nm nm)
                        ?session sess)
                    r3 = maeSetSpec(n2 tst ?gt gmidSatMargin ?session sess)
                    unless( r1 && r2 && r3
                        nErr = nErr + 1
                        printf("Fallo al anadir salidas/spec de %s en %s\n" nm tst))
                )
            )

            if( tests then
                maeSaveSetup(?session sess)
                printf("gm/Id: %d NMOS, %d PMOS | sesion %s | tests %L | spec vdsmargin > %s | %d fallos\n"
                    nN nP sess tests gmidSatMargin nErr)
            else
                printf("Ningun test de gmidTests existe en la sesion %s: %L\n"
                    sess maeGetSetup(?session sess))
            )
        )
    )
)

procedure( RegistrarMiMenuGmId(args)
    let( (itemMenu menuCustom win)
        win = if( windowp(args) then args else args->window )
        when( win && windowp(win)
            itemMenu = hiCreateMenuItem(
                ?name 'ItemGmId
                ?itemText "Generar"
                ?callback "GenerarMaestroGmId()"
            )
            menuCustom = hiCreatePulldownMenu(
                'MenuGmId
                "Crear expresiones"
                list(itemMenu)
            )
            hiInsertBannerMenu(win menuCustom length(hiGetBannerMenus(win)))
        )
    )
)

;; -------------------------------------------------------------------------
;; Chequeo periodico: anade el menu a cualquier esquematico abierto que
;; todavia no lo tenga (cubre ventanas ya abiertas y nuevas, sin depender
;; de que el trigger de apertura se dispare).
;; -------------------------------------------------------------------------
procedure( ChequearMenusGmId()
    errset(
        foreach( w hiGetWindowList()
            errset(
                when( windowp(w) && geGetEditCellView(w) &&
                      !member('MenuGmId hiGetBannerMenus(w))
                    RegistrarMiMenuGmId(list(nil 'window w))
                )
            )
        )
    )
    hiRegTimer("ChequearMenusGmId()" 1000)
)

;; Registro inmediato al cargar el plugin
ChequearMenusGmId()
